cmake_minimum_required(VERSION 3.22)
project(BasementAudio VERSION 0.1.0 LANGUAGES C CXX)
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.12 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
juce_add_console_app(BasementAudio PRODUCT_NAME "Basement Audio")
target_sources(BasementAudio PRIVATE Main.cpp Engine.cpp AudioCallback.cpp Rendering.cpp Midi.cpp Latency.cpp Editing.cpp Export.cpp Recording.cpp PluginHost.cpp DeviceCommands.cpp SessionCommands.cpp Commands.cpp)
target_compile_features(BasementAudio PRIVATE cxx_std_20)
target_compile_definitions(BasementAudio PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_PLUGINHOST_VST3=1 JUCE_MODAL_LOOPS_PERMITTED=1)
if(WIN32)
    target_compile_definitions(BasementAudio PRIVATE JUCE_ASIO=1)
endif()
target_link_libraries(BasementAudio PRIVATE juce::juce_audio_utils juce::juce_recommended_config_flags)
option(BASEMENT_TEST_PLUGIN "Build a private VST3 test fixture (not shipped)" ON)
if(BASEMENT_TEST_PLUGIN)
    foreach(fixture IN ITEMS TestInstrument TestDelay)
        if(fixture STREQUAL "TestInstrument")
            set(fixture_code Tsin)
        else()
            set(fixture_code Tsdl)
        endif()
        juce_add_plugin(${fixture} COMPANY_NAME "Basement Tests" BUNDLE_ID "studio.basementband.${fixture}" PLUGIN_MANUFACTURER_CODE Bsmt PLUGIN_CODE ${fixture_code} FORMATS VST3 PRODUCT_NAME "Basement ${fixture}" COPY_PLUGIN_AFTER_BUILD FALSE IS_SYNTH $<STREQUAL:${fixture},TestInstrument> NEEDS_MIDI_INPUT TRUE)
        if(fixture STREQUAL "TestInstrument")
            target_sources(${fixture} PRIVATE TestInstrument.cpp)
        else()
            target_sources(${fixture} PRIVATE TestGain.cpp)
            target_compile_definitions(${fixture} PRIVATE BASEMENT_TEST_DELAY=1)
        endif()
        target_compile_features(${fixture} PRIVATE cxx_std_20)
        target_compile_definitions(${fixture} PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
        target_link_libraries(${fixture} PRIVATE juce::juce_audio_utils juce::juce_recommended_config_flags)
    endforeach()
    juce_add_plugin(TestGain COMPANY_NAME "Basement Tests" BUNDLE_ID "studio.basementband.testgain" PLUGIN_MANUFACTURER_CODE Bsmt PLUGIN_CODE Tsgn FORMATS VST3 PRODUCT_NAME "Basement Test Gain" COPY_PLUGIN_AFTER_BUILD FALSE)
    target_sources(TestGain PRIVATE TestGain.cpp)
    target_compile_features(TestGain PRIVATE cxx_std_20)
    target_compile_definitions(TestGain PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
    target_link_libraries(TestGain PRIVATE juce::juce_audio_utils juce::juce_recommended_config_flags)
endif()
